void Trig_Example (void) {

double h, b, Ang =60;

/* Построение прямоугольного треугольника с гипотенузой L и углом Ang */

h = L * SinD (Ang); /* высота треугольника */

b = L * CosD (Ang); /* основание треугольника */

LineSeg ( 0, 0, 0, h, 1);

LineSeg ( 0, h, b, 0, 1);

LineSeg ( b, 0, 0, 0, 1);

/* Связь катетов */

h = b * TanD (Ang);

/* Обратное вычисление угла */

Ang = AtanD (h / b);

}; /* Trig_Example */
